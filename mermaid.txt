classDiagram
    class ZestDB {
        -LRUCache cache
        -IndexManager indexManager
        -StorageManager storageManager
        -Manifest manifest
        +get(key: String) Value
        +set(key: String, value: Value) void
        +delete(key: String) void
        -boot() void
    }

    class LRUCache {
        -Map~String, IndexEntry~ map
        -int capacity
        +get(key: String) IndexEntry
        +put(key: String, entry: IndexEntry) void
        +remove(key: String) void
    }

    class IndexManager {
        -String basePath
        +search(key: String) IndexEntry
        +update(key: String, entry: IndexEntry) void
        -getPartitionFile(key: String) File
        -binarySearch(file: File, key: String) IndexEntry
    }

    class StorageManager {
        -DataSegment currentSegment
        -List~DataSegment~ oldSegments
        +append(value: Value) IndexEntry
        +read(entry: IndexEntry) Value
        -createNewSegment() void
    }

    class DataSegment {
        -int segmentId
        -long currentOffset
        -boolean isFull
        +write(value: Value) long
        +read(offset: long, size: int) byte[]
    }

    class Compactor {
        +run() void
        -merge(seg1: DataSegment, seg2: DataSegment) DataSegment
        -isEntryValid(entry: IndexEntry) boolean
    }

    class IndexEntry {
        +String key
        +int segmentId
        +long offset
        +int size
        +boolean isTombstone
    }

    ZestDB --> LRUCache : "Consulte d'abord"
    ZestDB --> IndexManager : "Recherche si Cache Miss"
    ZestDB --> StorageManager : "Ecrit/Lit les données brutes"
    ZestDB --> Compactor : "Lance en arrière-plan"
    StorageManager "1" *-- "many" DataSegment : "Gère"
    IndexManager ..> IndexEntry : "Retourne"
    Compactor --> IndexManager : "Vérifie validité"
    Compactor --> StorageManager : "Nettoie"

--------------------------- Opération GET

sequenceDiagram
    participant Client
    participant ZestDB
    participant LRUCache
    participant IndexManager
    participant StorageManager

    Client->>ZestDB: get("Banane")
    ZestDB->>LRUCache: get("Banane")
    
    alt Dans le Cache (Hit)
        LRUCache-->>ZestDB: IndexEntry {seg: 1, off: 1024, size: 50}
    else Absent du Cache (Miss)
        ZestDB->>IndexManager: search("Banane")
        IndexManager->>IndexManager: Identifier partition (index_B.zst)
        IndexManager->>IndexManager: Binary Search sur disque
        IndexManager-->>ZestDB: IndexEntry {seg: 1, off: 1024, size: 50}
        ZestDB->>LRUCache: put("Banane", entry)
    end

    ZestDB->>StorageManager: read(entry)
    StorageManager->>StorageManager: Ouvrir Segment 1
    StorageManager->>StorageManager: Seek(1024)
    StorageManager-->>ZestDB: Valeur brute (byte[])
    ZestDB-->>Client: Valeur (JSON/Image/Texte)

--------------------------- Opération SET

sequenceDiagram
    participant Client
    participant ZestDB
    participant StorageManager
    participant IndexManager
    participant LRUCache

    Client->>ZestDB: set("Pomme", [Data])
    ZestDB->>StorageManager: append([Data])
    StorageManager->>StorageManager: Write at End of CurrentSegment
    StorageManager-->>ZestDB: Metadata (SegmentID, Offset, Size)
    
    par Mise à jour des index
        ZestDB->>IndexManager: update("Pomme", metadata)
        ZestDB->>LRUCache: put("Pomme", metadata)
    end
    
    ZestDB-->>Client: Success (OK)
